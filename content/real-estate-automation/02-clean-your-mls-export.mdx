```python
import pandas as pd

raw = pd.read_csv("leads-messy.csv")

# 1. Tidy the column names: lower case, no spaces. ('Lead ID' becomes lead_id.)
raw.columns = raw.columns.str.strip().str.lower().str.replace(" ", "_")

# 2. Drop rows that are completely empty.
clean = raw.dropna(how="all")

# 3. Keep one row per lead.            <-- your turn (blank 1)
clean = clean.drop_duplicates(subset=["____"])

# 4. Hard rule: never contact a do-not-contact lead.   <-- your turn (blank 2)
clean = clean[clean["status"] != "____"]

print(clean[["lead_id", "name", "status"]])
print(len(clean), "leads after cleaning")
```
